export interface CpuState {
  a: number; b: number; c: number; d: number; e: number;
  f: number; h: number; l: number; sp: number; pc: number;
  ime: number; eiDelay: number; halt: number; haltBug: number; stop: number;
}

export function initCpuState(): CpuState {
  return { a: 0x01, b: 0x00, c: 0x13, d: 0x00, e: 0xd8, f: 0xb0, h: 0x01, l: 0x4d, sp: 0xfffe, pc: 0x0100, ime: 0, eiDelay: 0, halt: 0, haltBug: 0, stop: 0 };
}

export interface CpuBus {
  read(addr: number): number;
  write(addr: number, v: number): void;
  tickM(): void;
  onOp(op: number): void;
  onCb(op: number): void;
  serviceInterrupt(): boolean;
  checkHaltExit(): boolean;
  cycles: number;
}

// flag helpers
function getZ(f: number): number { return (f >> 7) & 1; }
function setFlags(z: number, n: number, h: number, c: number): number { return (z << 7) | (n << 6) | (h << 5) | (c << 4); }

export function cpuStep(s: CpuState, bus: CpuBus): void {
  // EI delay handling: if eiDelay pending, count down
  // service interrupt first?
  // HALT behavior
  if (s.halt === 1) {
    // halt idle step: one M-cycle tick, check interrupt exit
    bus.tickM();
    if (bus.checkHaltExit()) {
      s.halt = 0;
    }
    // also try servicing if IME
    if (s.eiDelay > 0) {
      s.eiDelay--;
      if (s.eiDelay === 0) s.ime = 1;
    } else {
      if (bus.serviceInterrupt()) {
        s.halt = 0;
      }
    }
    return;
  }

  // EI delay: interrupts serviced after next instruction? Actually EI enables after 1 instruction delay.
  // Interrupt check happens before fetch each instruction, except during delay.
  if (s.eiDelay > 0) {
    s.eiDelay--;
    if (s.eiDelay === 0) s.ime = 1;
    // do not service this step? Actually after decrement to 0, IME set, but check next step.
    // Standard: EI then next instruction executes with IME=0, then IME=1. So don't service now.
    execOne(s, bus);
    return;
  }
  if (bus.serviceInterrupt()) {
    // serviced (clears halt too)
    s.halt = 0;
    return;
  }
  execOne(s, bus);
}

function fetchByte(s: CpuState, bus: CpuBus): number {
  let v: number;
  if (s.haltBug === 1) {
    s.haltBug = 0;
    v = bus.read(s.pc); // no increment
  } else {
    v = bus.read(s.pc);
    s.pc = (s.pc + 1) & 0xffff;
  }
  return v;
}

      bus.tickM(); break;

}

// EI delay semantics: we set 2, then cpuStep decrements at start of next steps. First step after EI: 2->1 executes instruction, second: 1->0 executes? That would be 2-instruction delay. Fix: set to 1 with special flag? Let's use: eiDelay=2 means: next instruction no service, after that enable. Our cpuStep: if eiDelay>0 { eiDelay--; if 0 ime=1; exec }. With eiDelay=2 set after EI: next step: 2->1, exec without ime; following: 1->0 sets ime=1 then exec (but ime already 1 during that exec — interrupt would be serviceable only after, that's fine since service check skipped this step). So delay is 1 full instruction. Good.

function pushAddr(s: CpuState, bus: CpuBus, v: number): void {
  bus.tickM();
  s.sp = (s.sp - 1) & 0xffff; bus.write(s.sp, (v >> 8) & 0xff);
  s.sp = (s.sp - 1) & 0xffff; bus.write(s.sp, v & 0xff);
}
function pushReg(s: CpuState, bus: CpuBus, hi: number, lo: number): void {
  s.sp = (s.sp - 1) & 0xffff; bus.write(s.sp, hi & 0xff);
  s.sp = (s.sp - 1) & 0xffff; bus.write(s.sp, lo & 0xff);
}

    } else if (grp === 1) { // RRC
      c = v & 1; r = ((c << 7) | (v >> 1)) & 0xff;

    } else if (grp === 2) { // RL

      s.f = setFlags(r === 0 ? 1 : 0, 0, 0, c);

      c = v & 1; const old = (s.f >> 4) & 1; r = ((old << 7) | (v >> 1)) & 0xff;
      s.f = setFlags(r === 0 ? 1 : 0, 0, 0, c);
    } else if (grp === 4) { // SLA
      c = (v >> 7) & 1; r = (v << 1) & 0xff;
      s.f = setFlags(r === 0 ? 1 : 0, 0, 0, c);
    } else if (grp === 5) { // SRA
      c = v & 1; r = ((v & 0x80) | (v >> 1)) & 0xff;

      s.f = setFlags(r === 0 ? 1 : 0, 0, 0, 0);

}

}
